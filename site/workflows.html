      const task = OdysseyWorkflows.mountTask(root, {
  adapter: {
    start: (state, { signal, onUpdate }) => host.start(state, signal),
    cancel: (state) => host.requestCancellation(state.operationId),
    refresh: (state) => host.read(state.operationId),
    retry: (state) => host.retry(state.operationId)
  }
});
// Success requires { state: 'success', operationId, receipt: { id } }.
// queued/running/cancelling remain nonterminal. Progress is source-only.
task.destroy(); // removes listeners and ignores late responses
